#+TITLE: Emacs Configuration for Productivity, Planning, "Getting Things Done" and Knowledge Management with Org Mode

This config is tangled and loaded by =init.el=. Self-contained logic lives in
=lisp/ps-*.el= modules and is ERT-tested; this file is configuration only.

If you are reading this to make it your own, the only section you usually need is
=* Settings= below — color theme, font, and every module knob live there.

* Bootstrap
** Package management
The first block carries the =-*- lexical-binding: t; -*-= cookie that the tangled
file needs. Do not move it.
#+begin_src emacs-lisp
  ;; -*- lexical-binding: t; -*-
  ;; Code above enables lexical scoping so lambdas can capture variables (needed for async code)

  ;; Keep the org-element cache globally OFF.  Parsing this very file with the
  ;; cache on can throw "Invalid search bound (wrong side of point)", so it must
  ;; stay off through startup.  The authoritative *tangle-time* guard lives in
  ;; init.el (set before `org-babel-load-file' tangles this file); this line is
  ;; the runtime companion — it only takes effect once config.el is built and
  ;; loaded.  Live Org *editing* buffers re-enable the cache locally in the Org
  ;; Mode section (see "Re-enable the org-element cache for live Org buffers"),
  ;; which keeps editing/scrolling fast without exposing the startup tangle.
  (setq org-element-use-cache nil)

  ;; Don't pop up the *Warnings* buffer when third-party packages byte/native-
  ;; compile on first install (e.g. guess-language emits a few harmless warnings
  ;; in its own source). Warnings are still logged to *Async-native-compile-log*,
  ;; and our own ps-* modules are kept warning-clean regardless.
  (setq native-comp-async-report-warnings-errors 'silent)

  ;; Same idea for the *synchronous* byte-compiler: package.el byte-compiles
  ;; every freshly installed package, and some (e.g. older MELPA packages) emit
  ;; harmless free-variable/obsolete-function warnings under a modern Emacs.
  ;; Suppress the *Compile-Log* buffer popping open for these; they're still
  ;; logged there for anyone who wants to check. Actual install/compile
  ;; failures raise a real Lisp error and surface in *Messages* regardless.
  (add-to-list 'warning-suppress-types '(bytecomp))

  ;; Suppress cosmetic byte-compiler "docstring wider than 80 characters"
  ;; warnings emitted when package.el synchronously byte-compiles a newly
  ;; installed third-party package (e.g. org-appear). Other warning categories
  ;; (free variables, obsolete functions, etc.) remain active.
  (setq byte-compile-warnings '(not docstrings))

  (require 'package)

  ;; Initialize package sources if needed
  (setq package-archives '(("melpa" . "https://melpa.org/packages/")
                           ("elpa" . "https://elpa.gnu.org/packages/")
                           ("nongnu" . "https://elpa.nongnu.org/nongnu/")))

  ;; Refresh package contents if needed
  (unless package-archive-contents
    (package-refresh-contents))

  ;; Install use-package if missing
  (unless (package-installed-p 'use-package)
    (package-install 'use-package))

  (require 'use-package)
  (setq use-package-always-ensure t)
#+end_src

** Auto-tangle this config on save
Saving =config.org= in a live session regenerates =config.el= immediately.
Because this tangle runs in a fully-initialized Emacs (org-element cache already
off, Org fully loaded) it is always clean — unlike the tangle =init.el= performs
at startup. It also leaves =config.el= newer than =config.org=, so the next
restart skips startup tangling entirely and just loads a known-good file.
#+begin_src emacs-lisp
  (defun ps/tangle-config-on-save ()
    "Tangle config.org -> config.el after saving config.org.
  Runs in the live session (org-element cache already off, Org fully
  loaded), so the tangle is clean — unlike the tangle init.el performs at
  startup. Regenerating config.el here also makes it newer than config.org,
  so the next restart skips startup tangling and just loads it."
    (let ((config (expand-file-name "config.org" user-emacs-directory)))
      (when (and buffer-file-name
                 (string= (file-truename buffer-file-name)
                          (file-truename config)))
        (let ((org-confirm-babel-evaluate nil))
          (org-babel-tangle-file
           config
           (expand-file-name "config.el" user-emacs-directory)
           "emacs-lisp")))))

  (add-hook 'after-save-hook #'ps/tangle-config-on-save)
#+end_src

** Org files (ps-org-files.el)
Which =.org= files the agenda, the conflict checker and the availability
computation see. The scan starts at your Org base directory and walks it
*recursively*, so any layout works — a flat folder, one level of topic
directories, or arbitrary nesting. Set =ps/org-files-root= only if you want to
scan a subdirectory rather than the whole base directory.

The two exclusion lists are regexps matched against a *name* (not a path):
directories matching the first are not descended into at all, and =.org= files
matching the second are dropped. The defaults skip dotted directories, the
journal, archives, and the two config files that live in the Org base directory.
#+begin_src emacs-lisp
  (setq ps/org-files-root nil)
  (setq ps/org-files-exclude-directories
        '("\\`\\." "\\`Journal\\'" "\\`Archive\\'"))
  (setq ps/org-files-exclude-files
        '("\\`init\\.org\\'" "\\`workspace\\.org\\'"))
#+end_src

** Blank line recovery (ps-blank-lines.el)
Beorg and Orgzly strip the blank lines out of org files when they save them.
=Recover blank lines from history= (F7) finds the version of each file that
still has them and reports what could be restored.  Nothing is written until
you review it: =d= in the report walks a file through ediff, and only what you
accept is saved.

Spacing is taken from history wherever history remembers it.  =unknown-spacing=
covers only the gaps it doesn't — a heading moved, or typed on the phone and
never seen intact: ='style= works those out from how the rest of your files are
spaced, ='leave-alone= proposes nothing there.  =match-floor= is how similar two
headings must be to count as the same one.  =side-by-side= puts your file and
the proposal in two columns during the review; nil stacks them.

The ancestor limits bound how far back each file's history is searched.  A
phone edit lands as a single commit and is usually noticed the same or the next
day, so a few commits and a day are enough; =max-days= of 0 means any age.  The
report's =-= and =+= step through =ancestor-steps=, and =c= / =y= set either
number outright.
#+begin_src emacs-lisp
  (setq ps/blank-lines-unknown-spacing     'style
        ps/blank-lines-match-floor          0.55
        ps/blank-lines-rule-min-samples     5
        ps/blank-lines-ancestor-max-commits 5
        ps/blank-lines-ancestor-max-days    1
        ps/blank-lines-ancestor-steps       '((1 . 1) (2 . 1) (3 . 1) (4 . 1) (5 . 1)
                                             (10 . 2) (20 . 3) (50 . 7) (100 . 14)
                                             (200 . 30) (500 . 90) (1000 . 365))
        ps/blank-lines-review-side-by-side  t)
#+end_src

** Scroll bars (ps-scrollbar.el)
Custom auto-hiding scroll-position indicator (the native macOS bar can't be
recoloured or made to auto-hide). A theme-coloured pill at the right edge shows
where you are; it appears when you scroll, idles for =ps/scrollbar-hide-delay=
seconds, then fades out over =ps/scrollbar-fade-duration= seconds (set to 0 for
an abrupt hide instead). =ps/scrollbar-width= is the pill width and
=ps/scrollbar-fringe-width= the reserved edge strip. =ps/scrollbar-show-on-hover=
also reveals it while the mouse is over its own track (not the whole window) --
on by default, since it is how you find the track to interact with.
Clicking anywhere in the track jumps there; disable with =ps/scrollbar-click-to-scroll= nil.
=ps/scrollbar-tick-interval= bounds how stale the thumb's position can look while
scrolling -- lower is snappier but costs more CPU (each tick calls =mouse-position=,
an NS cross-process call). That fast rate applies only while something is
happening; when idle the tick drops to =ps/scrollbar-idle-poll-interval= (a scroll
escalates it again immediately, so scroll-reveal is never delayed -- only
hover-reveal can take up to that long to notice the pointer). Losing OS focus
(e.g. minimizing Emacs) pauses the timer entirely; after
=ps/scrollbar-teardown-delay= seconds still unfocused, the remaining hooks/advice
are fully uninstalled too (reinstalled transparently on focus gain), so nothing in
this module runs at all during a long idle stretch.

To disable the custom scrollbar entirely (revert to native bars), set
=ps/scrollbar-enabled= to =nil= in =local.el=.
#+begin_src emacs-lisp
  (defvar ps/scrollbar-enabled t
    "Set to nil in local.el to disable the custom scrollbar entirely.")

  (setq ps/scrollbar-width             6
        ps/scrollbar-fringe-width     14
        ps/scrollbar-hide-delay       1.0
        ps/scrollbar-fade-duration    0.25
        ps/scrollbar-show-on-hover    t
        ps/scrollbar-tick-interval    0.15
        ps/scrollbar-idle-poll-interval 1.0
        ps/scrollbar-min-thumb-pixels 24
        ps/scrollbar-teardown-delay   (* 30 60))
#+end_src

** Material icons (ps-material-icons.el)
Category icons (in the agenda and the file tree) are drawn from the "Material
Symbols Outlined" font — no per-icon SVG files. See the README for installing
the font. Icon *size* auto-derives from your font height (so alignment stays
stable across fonts/OSes); tune =ps/material-icons-height-scale= once if needed.

Three maps decide which icon each tree entry gets: =category-map=
(=<Category>.org= → icon), =folder-map= (a *directory's own* icon, at any
depth), and =folder-contents-map= (one icon for every file under a folder).
All three are left *empty* here, so =config.org= alone renders files with the
generic =File= icon and directories with a plain folder. Real mappings live in
your Org folder's =workspace.org= (see "Workspace config" below), so your
personal categories never enter this public repo. Names come from
https://fonts.google.com/icons.
#+begin_src emacs-lisp
  (setq ps/material-icons-font-family "Material Symbols Outlined")
  (setq ps/material-icons-color       "#5f6368")
  (setq ps/material-icons-height      'auto)
  (setq ps/material-icons-height-scale 1.0)
  (setq ps/material-icons-category-map        nil)
  (setq ps/material-icons-folder-map          nil)
  (setq ps/material-icons-folder-contents-map nil)
#+end_src

** File Tree (ps-file-tree.el / ps-file-tree-icons.el)
Files/dirs whose name matches any regexp here are hidden from the file tree
(=init.org=, dotfiles, =elpa/=, byte-compiled files by default).

=ps/file-tree-root-mode= decides the tree's shape: =single= (the default) shows
your whole Org directory as one root, so files sitting directly in it are
visible and subdirectories nest to any depth; =subdirs= instead gives every
immediate subdirectory its own bold root, at the cost of hiding files that live
directly in the Org directory. With =ps/file-tree-hide-root= on, that single
root line isn't drawn at all — the tree simply starts at your folders and files
— and =ps/file-tree-root-name= (the root's label) goes unused.
=ps/file-tree-top-level-spacing= is the gap above each top-level folder (in
fractions of a line), and
=ps/file-tree-bold-top-level= draws those folders bold, so the tree still reads
as sections.

=ps/file-tree-order= sets the order entries appear in: names before the =:rest=
keyword are pinned to the top, names after it to the bottom, and everything
unlisted fills the middle alphabetically. Leave it =nil= for plain alphabetical
order. Real orderings usually belong in your =workspace.org=, next to the icon
maps. =ps/file-tree-show-toggle-button= puts a collapse/expand-all button and a
refresh button in the tree's top-right corner.

=ps/file-tree-follow-current-file= keeps the tree's highlight on the file you're
editing. It never expands anything and never moves the view: if the file isn't
already on screen in the tree, the highlight is cleared instead.

Category icons
come from the Material Symbols font (see above); =ps/file-tree-icon-ascent= sets
the vertical alignment of every file-tree icon (root projects and files alike).
If the font is unavailable, the tree falls back to the =folder=/=folder_open=
/=draft= SVGs in =ps/file-tree-icon-fallback-dir=. Set =ps/file-tree-use-custom-icons=
to =nil= to use treemacs's generic icons instead.
#+begin_src emacs-lisp
  (setq ps/file-tree-ignored-files
        '("\\`init\\.org\\'" "\\`workspace\\.org\\'"
          "\\`AGENTS\\.md\\'" "\\`CLAUDE\\.md\\'"
          "\\`\\." "\\`elpa\\'" "\\.elc\\'"))

  (setq ps/file-tree-root-mode 'single)
  (setq ps/file-tree-root-name nil)
  (setq ps/file-tree-hide-root t)
  (setq ps/file-tree-top-level-spacing 0.35)
  (setq ps/file-tree-bold-top-level t)

  (setq ps/file-tree-order nil)
  (setq ps/file-tree-show-toggle-button t)

  (setq ps/file-tree-follow-current-file t)
  (setq ps/file-tree-follow-delay 0.2)

  (setq ps/file-tree-use-custom-icons t)
  (setq ps/file-tree-name-spacing 0.5)
  (setq ps/file-tree-icon-ascent 80)
  (setq ps/file-tree-icon-fallback-dir
        (expand-file-name "icons" user-emacs-directory))
#+end_src

*** File sets
Named filters for which files the tree shows — by default this is purely
a display filter, and the agenda and everything else still see every
file. Each set has =:include= and/or =:exclude= regexps, matched as
substrings of each file's absolute path. "All" (the default) shows
everything, and is also the fallback used if =ps/file-tree-current-set=
ever names a set that no longer exists. The active set is shown and
switched from the file tree's mode line, and persists across restarts.
#+begin_src emacs-lisp
  (setq ps/file-tree-file-sets
        '(("All" . (:include nil :exclude nil))
          ;; ;; Examples:
          ;; ("Work" . (:include ("/Work/"
          ;;                      "/Current/")
          ;;            :exclude nil))
          ;; ("No Sensitive" . (:include nil
          ;;                     :exclude ("/Admin/Financial\\.org\\'"
          ;;                               "/Work/Interviews\\.org\\'")))
          ))
#+end_src

** Agenda emoji matcher (ps-agenda-emoji.el)
The agenda appends a semantic emoji to each task. Matches come from
=scripts/org_emoji_matcher.py= and are kept in a persistent on-disk cache, so a
warm agenda render reuses cached emojis and never spawns Python.

Set =ps/agenda-emoji-enabled= to nil to turn the agenda emojis off entirely.

(and, if you like, set =EMOJI_MODEL= / =EMOJI_THRESHOLD= for the script) to
invalidate the cache after changing the matcher. =right-margin= is how many
columns from the right window edge the emoji column starts.

  (setq ps/agenda-emoji-enabled      t   ; set to nil to disable agenda emojis

        ps/agenda-emoji-cache-file
        (expand-file-name "emoji_matcher/task_cache.json"
                          (expand-file-name ".cache" "~"))
        ps/agenda-emoji-cache-tag    "bge-small-en-v1.5@0.50"
        ps/agenda-emoji-right-margin 3
        ;; Emoji fonts are already scaled to the text line box by

        ps/agenda-emoji-face         '(:height 1.0))

** Agenda layout (ps-agenda-layout.el)
The agenda task lines are re-laid into aligned columns:
=[category] [STATE] [PRIORITY] [emoji]  Title… [tags]   [when]=. State, priority
and the relative date ("11d ago") are drawn as rounded pill badges; titles line
up across every section and are truncated (full text shown on hover) when they
would overflow.

- =category-display=: =icon= (default), =name=, =both=, or =none=.
- =schedule-style=: =grid= keeps Org's native time ruler; =compact= hides the
  empty grid rows and lays timed events out like the other sections. Switch and
  press =g= in the agenda to compare.
- =state-labels=: remap a keyword to a shorter label, e.g. ="WAIT" → "W"= (the
  actual TODO keyword is unchanged). Rarely needed now that all keywords are ≤4
  chars. Empty = show keywords verbatim.
- =state-cols= is auto-computed from the longest keyword + 2 padding spaces when
  left at =nil=; set to an integer to override. =priority-cols= is =4= (covers
  =" #A "=). State and priority badges use =org-modern= faces, so their colours
  follow the active theme automatically — no per-theme colour overrides needed.
#+begin_src emacs-lisp
  (setq ps/agenda-layout-enabled        t
        ps/agenda-layout-category-display 'icon   ; icon | name | both | none
        ps/agenda-layout-schedule-style   'grid   ; grid | compact
        ps/agenda-layout-state-labels     nil     ; e.g. '(("WAIT" . "W"))
        ps/agenda-layout-truncate         t
        ps/agenda-layout-priority-cols    3
        ps/agenda-layout-emoji-cols       2
        ps/agenda-layout-category-cols    2
        ps/agenda-layout-right-margin-cols 1)
#+end_src

** Schedule view (ps-schedule-view.el)
Replaces the native time-grid rendering for the Schedule section with two
styles, switchable from the =Productivity → Schedule View= menu:

- =Timeline= — shows the full time axis: a tick row for each grid interval

** Typo / spell checking (ps-typo.el)
A quiet, multilingual typo checker for Org buffers. A word is underlined only
when *no* configured language recognises it, so terms valid in any of your
languages (e.g. Spanish =calor=, Cyrillic words) are never flagged. Code, links
and markup are skipped automatically. Correct a word with =M-$=.

=ps/typo-languages= is the union of accepted languages — extend it for every
language you write in (each needs an =enchant= dictionary; see README.md).
=ps/typo-near-miss-gate= is an extra-precision mode (only flag words that look
like a near miss of a real word); it is off by default so you can experiment.
=ps/typo-order-suggestions-by-language= lists fixes for the locally-detected
language first; set it to nil to disable. =ps/typo-underline-color= is the wavy
underline colour (=auto= derives a muted colour from the theme's =shadow= face).
=ps/typo-inhibit-predicate= excludes the read-only documentation reading view
(=ps/help--doc-buffer-p=, defined later in this file) — those pages cannot be
edited, so spell-checking them is just noise.
#+begin_src emacs-lisp
  (setq ps/typo-languages       '("en_US" "ru_RU" "es_ES")
        ps/typo-min-length      3
        ps/typo-near-miss-gate  nil   ; t = only flag near-misses of real words
        ps/typo-near-miss-max-distance 2
        ps/typo-order-suggestions-by-language t
        ;; Solarized keeps a soft grey; other themes derive one from `shadow'.

** Workspace config (per-Org-folder overrides)
A small =workspace.org= in your Org base directory (beside =init.org=) holds
settings specific to *this data folder* — personal category→icon mappings and
file sets — overriding the (intentionally empty) defaults above. It is loaded
here, after those defaults but before icons are applied, so its overrides take
effect. Keeping personal data there (not in =config.org=) means pulling upstream
changes never conflicts. It is loaded artifact-free: its elisp is tangled to a
throwaway file in the system temp dir, never inside your Org folder.
#+begin_src emacs-lisp
  (defun ps/workspace-config-file ()
    "Return the path to this Org folder's workspace.org."
    (expand-file-name "workspace.org" my-org-base-directory))

  (defun ps/load-workspace-config ()
    "Load emacs-lisp from workspace.org, if it exists.
Tangles its blocks to a temp .el in the system temp dir (never the Org dir),
loads it, then deletes it. Silently does nothing when the file is absent."
    (let ((file (ps/workspace-config-file)))
      (when (file-exists-p file)
        (require 'ob-tangle)
        (let ((el (make-temp-file "ps-workspace-" nil ".el")))
          (unwind-protect
              (let ((org-confirm-babel-evaluate nil))
                (org-babel-tangle-file file el "emacs-lisp")
                (load el nil t))
            (when (file-exists-p el) (delete-file el)))))))

  (defun ps/icons-reapply ()
    "Re-apply agenda and file-tree icons from the current maps."
    (ps/agenda-icons-apply)
    (when (and ps/file-tree-use-custom-icons (featurep 'treemacs))
      (ps/file-tree-icons-apply my-org-base-directory)
      (when (fboundp 'ps/file-tree-refresh) (ps/file-tree-refresh))))

  (defun ps/open-workspace-config ()
    "Open this Org folder's workspace.org (a fresh buffer if it doesn't exist)."
    (interactive)
    (find-file (ps/workspace-config-file)))

  (defun ps/reload-workspace-config ()
    "Reload workspace.org and re-apply icons, file sets and situations."
    (interactive)
    (ps/load-workspace-config)
    (ps/icons-reapply)
    ;; Re-derive org-tag-alist, the agenda commands and the C-c p S keymap from
    ;; the (possibly edited) ps/context-tags and ps/situations.
    (when (fboundp 'ps/situations-apply) (ps/situations-apply))
    (when (fboundp 'ps/agenda-files-refresh) (ps/agenda-files-refresh))
    (message "Workspace config reloaded"))

  ;; Apply this folder's overrides now, before icons are built below.
  (ps/load-workspace-config)
#+end_src

** Claude Code (ps-claude.el)
=ps/claude-window-width= is the side-window width in columns (the upstream
default of 100 is quite wide). =ps/claude-resize-debounce-delay= controls how
long Emacs waits, after a window resize settles, before re-syncing the
terminal's dimensions with the =claude= process — this fixes output that
otherwise renders into only part of the window (or looks garbled) until
another resize forces a redraw. =ps/claude-reanchor-delay= is how long to wait
before a second pass that re-positions the window once Claude has actually
repainted. =ps/claude-resize-throttle-interval= limits how often the terminal
reflows during a burst of resize events (e.g. dragging the divider) — the
reflow is what makes the panel blink, so a larger value means less flicker
but coarser tracking during the drag. =ps/claude-resize-burst-gap= is how
close together resize events must arrive to count as the same burst;
anything farther apart is treated as an isolated resize and always reflows
immediately, regardless of the throttle. =ps/claude-drag-clip-max-duration=
is a safety ceiling on how long a burst can keep the panel's lines clipped
(rather than wrapped) before it's forced back, in case a burst never
settles. =ps/claude-no-exit-prompt= lets Emacs quit without asking to confirm
killing a running session. =ps/claude-debug-resize= appends troubleshooting
detail to =ps/claude-debug-resize-file=; leave it off otherwise.
#+begin_src emacs-lisp
  (setq ps/claude-window-width 90
        ps/claude-resize-debounce-delay 0.3
        ps/claude-reanchor-delay 0.2
        ps/claude-resize-throttle-interval 0.25
        ps/claude-resize-burst-gap 0.15
        ps/claude-drag-clip-max-duration 2.0
        ps/claude-no-exit-prompt t
        ps/claude-debug-resize nil)
#+end_src

Scrolling past the start or end of a buffer normally echoes "Beginning of
buffer"/"End of buffer" -- silence just that, leaving every other message
intact. =pixel-scroll-precision= (the active backend here) catches that
condition internally and calls =message= directly rather than signalling an
error up to the command loop, so it has to be caught at the =message= level,
not via =command-error-function= (kept below too, for the ordinary
keyboard-scroll case where the error does reach the top level).
#+begin_src emacs-lisp
  (defun ps/quiet-buffer-bounds-error (data context caller)
    (unless (memq (car data) '(beginning-of-buffer end-of-buffer))
      (command-error-default-function data context caller)))
  (setq command-error-function #'ps/quiet-buffer-bounds-error)

  (define-advice message (:filter-args (args) ps/quiet-buffer-bounds-message)
    (if (member (car args) '("Beginning of buffer" "End of buffer"))
        (list nil)
      args))
#+end_src

Fast trackpad scrolling synthesizes =double-wheel-up=/=triple-wheel-up= events
(and the other three directions) the same way rapid clicks become
=double-mouse-1=; neither =mwheel-scroll= nor =pixel-scroll-precision= bind
those, so they flash "is undefined" in the echo area even though both already
ignore the click-multiplicity once actually invoked. Re-dispatch them as their
plain equivalent instead of leaving them unbound. Separately, a wheel event
*or click* over pixels that belong to no window at all (the sliver between a
window's edge and the literal frame border) is reported as the two-event key
sequence =[nil wheel-up]=/=[nil down-mouse-1]= -- shown as "<nil> <wheel-up>"
etc. -- which nothing binds; silence those outright since there is no window
underneath to act on.
#+begin_src emacs-lisp
  (defun ps/wheel-strip-multiplicity (event)
    "Re-dispatch a double/triple wheel EVENT as its plain (single) equivalent."
    (interactive "e")
    (let* ((mods (delq 'double (delq 'triple (event-modifiers event))))
           (basic (event-basic-type event))
           (type (if mods (event-convert-list (append mods (list basic))) basic))
           (cmd (key-binding (vector type) t nil (event-start event))))
      (when (commandp cmd)
        (let ((last-input-event (cons type (cdr event))))
          (call-interactively cmd)))))

  (dolist (dir '(wheel-up wheel-down wheel-left wheel-right))
    (dolist (mult '(double triple))
      (let ((type (event-convert-list (list mult dir))))
        (dolist (prefix '(right-fringe left-fringe mode-line header-line
                           vertical-scroll-bar tool-bar left-margin right-margin))
          (global-set-key (vector prefix type) #'ps/wheel-strip-multiplicity))))
    (dolist (mult '(nil double triple))
      (let ((type (if mult (event-convert-list (list mult dir)) dir)))
        (global-set-key (vector nil type) #'ignore))))

  ;; Same no-man's-land sliver, but for clicks: =[nil down-mouse-1]= ("<nil>
  ;; <mouse-1>" in the echo area) falls through to Emacs's default click
  ;; handling against whatever happens to be in that screen position, which
  ;; can visibly corrupt unrelated faces elsewhere in the frame for a moment
  ;; (e.g. org-modern TODO/tag pills rendering outline-only, the mode-line
  ;; looking inactive) until something else forces a clean redisplay.
  (dolist (type '(down-mouse-1 mouse-1 drag-mouse-1))
    (global-set-key (vector nil type) #'ignore))

  ;; Wheel events in the vertical-line area (window divider) are also
  ;; unbound and produce echo-area noise when the mouse is near the scrollbar
  ;; track adjacent to a split.  Single variants forward to the left window
  ;; (`posn-window' for vertical-line events is the left window whose right
  ;; fringe IS the scrollbar track); double/triple strip multiplicity as usual.
  (defun ps/vertical-line-forward-wheel (event)
    "Forward a wheel EVENT in the vertical-line area to the adjacent window."
    (interactive "e")
    (let* ((win (posn-window (event-start event)))
           (basic (event-basic-type event))
           (cmd (when (window-live-p win)
                  (key-binding (vector basic) t nil (event-start event)))))
      (when (commandp cmd)
        (setcar (nth 1 event) win)
        (let ((last-input-event (cons basic (cdr event))))
          (call-interactively cmd)))))

  (dolist (dir '(wheel-up wheel-down wheel-left wheel-right))
    (global-set-key (vector 'vertical-line dir) #'ps/vertical-line-forward-wheel)
    (dolist (mult '(double triple))
      (global-set-key (vector 'vertical-line (event-convert-list (list mult dir)))
                      #'ps/wheel-strip-multiplicity)))

  ;; Silence echo-area messages triggered by hovering over window-chrome
  ;; areas: "drag-mouse-1: resize window" (vertical-line divider) and
  ;; "mouse-1: select window..." (mode-line).  Both come through
  ;; `show-help-function'; `add-function :before-while' returns nil (blocking
  ;; the main function) only for those specific strings, letting all other
  ;; help-echo messages through normally.
  (add-function :before-while show-help-function
    (lambda (msg)
      (or (null msg)
          (not (string-match-p
                "\\(?:drag-mouse-[0-9]+\\|mouse-[0-9]+.*\\(?:window\\|frame\\)\\)" msg))))
    '((name . ps/silence-window-chrome-hints)))
#+end_src

** Line numbers
#+begin_src emacs-lisp
  ;; Show column number in mode line
  (column-number-mode)

  ;; Reserve a fixed 2-column gutter.
  ;;
  ;; Left nil, Emacs recomputes the width on every redisplay as
  ;; log10(MAX-LNUM)+1, where MAX-LNUM is the line at window start plus the
  ;; window's `desired_matrix->nrows'. That row count is
  ;; (window-pixel-height + |vscroll|) / FRAME_SMALLEST_FONT_HEIGHT, and
  ;; FRAME_SMALLEST_FONT_HEIGHT is hardcoded to 1 on this build (matrices are
  ;; deliberately over-allocated) — so nrows is the window's height in *pixels*,
  ;; not lines, and MAX-LNUM near the top of a buffer is ~1000 regardless of how
  ;; long the file actually is. Sitting on that power-of-ten boundary, the
  ;; sub-line `vscroll' that pixel-precise scrolling sets during a gesture is
  ;; enough to flip the width 4 -> 3 -> 4; since line numbers are right-aligned
  ;; within the gutter, the digits and the text jump left and back together.
  ;; 4 is the width the estimate settles on anyway once you scroll past the
  ;; boundary, so pinning it removes the flicker without preventing genuine
  ;; growth (files past ~9000 lines still get a 5th column). `setq-default' is
  ;; required: the variable is permanently buffer-local.
  (setq-default display-line-numbers-width 2)

  ;; Display line numbers globally
  (global-display-line-numbers-mode t)

  ;; Major modes that must NEVER show line numbers. Stock Emacs has no built-in
  ;; exempt list for `global-display-line-numbers-mode' — its turn-on function
  ;; only skips the minibuffer and otherwise enables numbers in every buffer.
  ;; Disabling the minor mode from a mode hook is fragile: the global mode
  ;; re-evaluates buffers on major-mode changes and can re-enable numbers when a
  ;; buffer is rebuilt, which is how the File Tree (treemacs) intermittently
  ;; ended up with line numbers (its buffer is recreated on toggle / visibility-
  ;; set switches). Instead, refuse to turn numbers on for these modes at the
  ;; source, so the decision holds every time the global mode evaluates a buffer.
  (defvar ps/display-line-numbers-exempt-modes
    '(treemacs-mode term-mode shell-mode eshell-mode org-agenda-mode eat-mode)
    "Major modes for which global line numbers must never be enabled.")

  (advice-add 'display-line-numbers--turn-on :before-while
              (lambda (&rest _)
                (not (apply #'derived-mode-p ps/display-line-numbers-exempt-modes)))
              '((name . ps/display-line-numbers-exempt)))
#+end_src

** Color theme
Loads =ps/color-theme= from Settings, auto-installing the package that provides it
when needed. The Solarized-specific tuning below only runs for =solarized-*=
themes; any other theme loads cleanly without it.
#+begin_src emacs-lisp
  ;; Theme-name prefix -> package that provides it. Built-in themes (e.g. modus-*
  ;; on Emacs 28+) install only if not already available, so listing them is harmless.
  (defvar ps/color-theme-packages
    '(("modus-"      . modus-themes)     ; accessible "default" style (built-in 28+)
      ("ef-"         . ef-themes)        ; colorful + legible (GNU ELPA)
      ("standard-"   . standard-themes)  ; faithful-to-default-Emacs, refined (GNU ELPA)
      ("doric-"      . doric-themes)     ; minimalist (GNU ELPA)
      ("doom-"       . doom-themes)      ; the popular megapack (MELPA)
      ("batppuccin-" . batppuccin)       ; Catppuccin, as 4 proper themes (MELPA)
      ("tokyo-night" . tokyo-night)      ; Tokyo Night, 4 variants (MELPA)
      ("gruvbox"     . gruvbox-theme)    ; retro groove (NonGNU ELPA / MELPA)
      ("nord"        . nord-theme)       ; arctic palette (MELPA)
      ("solarized-"  . solarized-theme)) ; default theme, kept for back-compat
    "Alist of (THEME-NAME-PREFIX . PACKAGE) for auto-installing theme packages.")

  (defun ps/load-color-theme (theme)
    "Install (if needed) and load THEME, applying Solarized tuning when relevant."
    (let* ((name (symbol-name theme))
           (pkg (cdr (seq-find (lambda (c) (string-prefix-p (car c) name))
                               ps/color-theme-packages))))
      ;; Pull in the providing package unless THEME is already available.
      (when (and pkg (not (memq theme (custom-available-themes))))
        (unless (package-installed-p pkg)
          ;; Always refresh: the cached archive contents may predate this
          ;; package's addition to MELPA/GNU ELPA, or reference a since-pruned
          ;; tarball version, both of which surface as "package unavailable".
          (package-refresh-contents)
          (package-install pkg))
        (require pkg nil t))
      ;; Solarized-only tuning (these variables don't exist for other themes).
      (when (string-prefix-p "solarized-" name)
        (setq
         solarized-distinct-fringe-background nil  ; fringe blends with background
         solarized-use-variable-pitch nil          ; keep one font for headings/titles
         solarized-high-contrast-mode-line nil     ; low-contrast modeline
         solarized-use-less-bold t                 ; less bolding
         solarized-use-more-italic t               ; more italics
         solarized-emphasize-indicators nil        ; muted git-gutter/flycheck indicators
         solarized-scale-org-headlines nil         ; don't resize org headlines
         ;; Avoid all font-size changes
         solarized-height-minus-1 1.0
         solarized-height-plus-1 1.0
         solarized-height-plus-2 1.0
         solarized-height-plus-3 1.0
         solarized-height-plus-4 1.0))
      (load-theme theme t)))

  (ps/load-color-theme ps/color-theme)

  ;; On the very first frame, Emacs may not yet know the display supports 256+
  ;; colors, so a theme's GUI-specific `default' face spec (e.g. doom themes'
  ;; `min-colors 256' background) can fail to apply, leaving the frame's
  ;; previous/default background behind. Reloading once the initial frame is
  ;; fully set up fixes the background without affecting already-correct themes.

    ;; displayed.  ":scheduled today" matches both timed and untimed

        (:name "Scheduled earlier:" :scheduled past    :order 5)
        (:name "Due soon:"          :deadline future   :order 6))
      "Date-bound super-agenda groups for single-day agenda views.")

    ;; agenda (C-c a a) and clicking a timestamp in an org file (which both call
    ;; `org-agenda-list' with no group override) render through the same pipeline
    ;; instead of a raw, ungrouped list.  Views that want a flat list bind this
    ;; to nil locally: `ps/show-tasks' (Tasks) and the Calendar's multi-day
    ;; branch (which relies on Org's native per-day headers instead).

                          (org-deadline-warning-days 0)

                          ;; diary sexps and the blocked-task computation Org
                          ;; would otherwise do for every entry.  This roughly

                          (org-agenda-entry-types '(:deadline :scheduled :timestamp))

  ;; Master switches org-appear toggles dynamically; off by default in Emacs,

** Behavior
*** Auto-save org buffers + auto-revert on disk change
#+begin_src emacs-lisp
  ;; Auto-save all Org buffers
  ;; https://christiantietze.de/posts/2019/03/sync-emacs-org-files/
  ;; Save quietly: `org-save-all-org-buffers' echoes "Saving all Org
  ;; buffers... done" on every auto-save, which both flashes in the echo
  ;; area and piles up in *Messages* when Emacs runs for days.  Binding
  ;; `inhibit-message' suppresses the echo-area flash and `message-log-max'
  ;; nil keeps it out of *Messages*; the save itself is unchanged.
  (defun ps/org-save-all-org-buffers-quietly ()
    "Save all Org buffers without echoing/logging the save message."
    (let ((inhibit-message t) (message-log-max nil))
      (org-save-all-org-buffers)))
  (add-hook 'auto-save-hook #'ps/org-save-all-org-buffers-quietly)

  ;; Obsidian-like persistence. The idle auto-save above keeps files current
  ;; while editing; these two close the remaining gaps so the mode line never
  ;; needs a "modified" indicator:
  ;;   - save the moment Emacs loses focus (e.g. switching to another app,
  ;;     which is also how Claude Code picks up the latest file content), and
  ;;   - save just before exit, so quitting never prompts "Save buffer?".
  (defcustom ps/org-save-on-focus-loss-idle-threshold 300
    "Seconds since the last command above which a focus-loss save is skipped.
The org files here live on a Dropbox-backed CloudStorage mount; a save
triggered long after Emacs was last actually used (e.g. a focus event
generated around system sleep/wake) is exactly when that mount is most
likely to be slow to respond -- and is not a deliberate hand-off to another
app, since nothing has happened in Emacs for a while anyway. Skipping it
there avoids blocking all of Emacs on that write; the next real save (the
periodic auto-save-hook one above, or a later focus-loss save) still covers
it."
    :type 'number)

  (defvar ps/org-save--last-command-time (float-time)
    "`float-time' of the most recent command Emacs processed.
Compared against `ps/org-save-on-focus-loss-idle-threshold' by
`ps/org-save-on-focus-loss' -- more direct than `current-idle-time', which
is not reliably meaningful from inside a focus-change callback.")
  (add-hook 'post-command-hook
            (lambda () (setq ps/org-save--last-command-time (float-time))))

  (defun ps/org-save--any-org-buffer-modified-p ()
    "Non-nil if any live Org buffer has unsaved changes."
    (seq-some (lambda (b)
                (and (buffer-modified-p b)
                     (with-current-buffer b (derived-mode-p 'org-mode))))
              (buffer-list)))

  (defun ps/org-save-on-focus-loss ()
    "Save all Org buffers when Emacs loses focus.
Skipped outright when nothing is modified, or when Emacs has been idle
longer than `ps/org-save-on-focus-loss-idle-threshold' (see its docstring).
The save itself is bounded by a timeout and never lets an error escape, so a
slow or failing save cannot take the rest of `after-focus-change-function'
-- e.g. `ps/scrollbar--on-focus-change' -- down with it."
    (unless (frame-focus-state)
      (when (and (ps/org-save--any-org-buffer-modified-p)
                 (< (- (float-time) ps/org-save--last-command-time)
                    ps/org-save-on-focus-loss-idle-threshold))
        (with-timeout (5 (message "ps/org-save-on-focus-loss: timed out, skipped"))
          (condition-case err
              (ps/org-save-all-org-buffers-quietly)
            (error (message "ps/org-save-on-focus-loss: %S" err)))))))
  (add-function :after after-focus-change-function #'ps/org-save-on-focus-loss)
  (advice-add 'save-buffers-kill-emacs :before
              (lambda (&rest _) (ps/org-save-all-org-buffers-quietly)))

  ;; Automatically revert buffers when files change on disk.
  ;; Rely on file-system notifications instead of the default ~5s polling
  ;; timer. On the macOS (NS) build, a redisplay forced from inside a timer
  ;; while Emacs is idle can re-enter AppKit's event loop and hard-freeze the
  ;; whole UI (no keyboard/mouse input). Notification-based reverts avoid that
  ;; recurring timer, and silencing revert output avoids extra echo-area
  ;; redisplay.
  (setq auto-revert-avoid-polling t
        auto-revert-verbose nil
        auto-revert-check-vc-info nil)
  (global-auto-revert-mode t)

  ;; A file replaced on disk by Dropbox (an edit made in a mobile Org app)
  ;; can arrive as one notification carrying several actions, and the action
  ;; that stops the watch leaves the ones behind it calling a callback that
  ;; is already gone -- "file-notify--call-handler: Symbol's function
  ;; definition is void: nil" in *Messages*, right after the revert.  See
  ;; ps-file-notify.el.
  (ps/file-notify-setup)

  ;; `global-auto-revert-mode' already rereads unmodified buffers when the
  ;; file changes on disk, but it waits for a filesystem notification (see
  ;; `auto-revert-avoid-polling' above).  Anything that visits the file inside
  ;; that window -- e.g. Claude Code re-opening a file it just wrote -- hits
  ;; `find-file-noselect's "Reread from disk?  (yes or no)" prompt instead.
  ;; Answer it the same way auto-revert would.  The empty regexp matches every
  ;; file, and only ever applies to *unmodified* buffers: one with unsaved
  ;; edits still prompts ("Discard your edits?"), so real conflicts are never
  ;; silently lost.  Trade-off: the silent reread echoes "Reverting file
  ;; ...done" instead, which is transient but does land in *Messages*.
  (setq revert-without-query '(""))
#+end_src

** =treemacs= file tree
A tree-style sidebar for browsing the Org GTD tree.
The ignore list and category icons live in =lisp/ps-file-tree.el= and
=lisp/ps-file-tree-icons.el=.
#+begin_src emacs-lisp
  (use-package treemacs
    :defer t
    :commands (treemacs treemacs-select-window)
    :config
    (ps/file-tree-setup-ignore)
    (ps/file-tree--ensure-valid-set)
    (if ps/file-tree-use-custom-icons
        (ps/file-tree-icons-apply my-org-base-directory)
      (treemacs-load-theme "Default"))

    ;; Replace the file tree's mode line entirely with the active file-set
    ;; indicator; click it to switch sets.
    (setq treemacs-user-mode-line-format '(:eval (ps/file-tree--modeline)))

    ;; A collapse/expand-all button in the top-right corner. The header line is
    ;; the only spot up there that stays put while the tree scrolls.
    (setq treemacs-user-header-line-format '(:eval (ps/file-tree--header-line)))

    ;; `treemacs-sorting' takes a predicate over absolute paths, which is how
    ;; `ps/file-tree-order' gets to place folders explicitly. With that unset
    ;; this sorts alphabetically, exactly like treemacs's own default.
    (setq treemacs-sorting #'ps/file-tree-sort-predicate)

    ;; Root project labels should look like ordinary file-tree entries (not
    ;; underlined or oversized like the default hyperlink-style root face),
    ;; but bold to stand out as section headers. Only visible with
    ;; `ps/file-tree-hide-root' off, or in `subdirs' root mode; the default
    ;; layout gets its section headers from `ps/file-tree-bold-top-level'.
    (set-face-attribute 'treemacs-root-face nil
      :inherit 'treemacs-file-face
      :height 1.0
      :weight 'bold
      :underline nil)

    ;; Folders read as the same kind of thing as the files under them, so give
    ;; them the file colour rather than treemacs's accent blue. Clearing the
    ;; theme's own :foreground is what lets the :inherit through. Top-level
    ;; folders still stand out, via `ps/file-tree-bold-top-level'.
    (set-face-attribute 'treemacs-directory-face nil
      :inherit 'treemacs-file-face
      :foreground 'unspecified)

    ;; Narrower than treemacs's 35-column default.
    (setq treemacs-width 25)

    ;; Keep treemacs's own width lock off: it works by setting
    ;; `window-size-fixed', which also blocks dragging the divider. The width
    ;; is held steady across frame resizes by `ps/file-tree-width-setup'
    ;; instead -- see design-docs/file-view-resizing.md.
    (setq treemacs-width-is-initially-locked nil)
    (ps/file-tree-width-setup)

    ;; More compressed indentation per level (default is 2).
    (setq treemacs-indentation 1)

    ;; Hide the .org extension and add breathing room after icons.
    (setq treemacs-file-name-transformer #'ps/file-tree-transform-file-name)
    (setq treemacs-directory-name-transformer #'ps/file-tree-transform-dir-name)

    ;; treemacs's own follow-mode expands directories and re-roots the view to
    ;; reveal the current file. Ours only moves the highlight, and only when the
    ;; file is already on screen, so the tree never shifts under you.
    (treemacs-follow-mode -1)
    (treemacs-tag-follow-mode -1)
    (ps/file-tree-follow-setup)

    ;; Hidden root line, section gaps and bold top-level folders. treemacs
    ;; re-renders through several paths that report no hook, so this installs
    ;; the handful of hooks needed to re-apply the styling after each one.
    (ps/file-tree-decorations-setup)

    ;; This is a curated org-category navigator, not a code-project view, so
    ;; don't tint files by git status (a modified file would otherwise show in
    ;; treemacs-git-modified-face, e.g. blue, hiding the category styling).
    (when (fboundp 'treemacs-git-mode) (treemacs-git-mode -1))

    ;; treemacs already turns this on by default when the package loads (see
    ;; `treemacs-only-during-init' in treemacs-filewatch-mode.el) -- stated
    ;; explicitly here, like the two toggles above, so it isn't silently
    ;; relying on that default. Watches every expanded directory, but only
    ;; reacts to whole files/folders appearing or disappearing there -- a
    ;; plain content change is a "changed" event, and treemacs drops those
    ;; unless `treemacs-git-mode' is on (see `treemacs--is-event-relevant?'
    ;; in treemacs-filewatch-mode.el), which this config deliberately keeps
    ;; off. In practice that covers a folder appearing or being removed; a
    ;; single new file landing inside an already-open folder often does not
    ;; fire a distinct enough event to be picked up. The more reliable
    ;; automatic path is `ps/git-sync--run's success sentinel, which calls
    ;; `ps/file-tree-refresh' after any pull that actually brought something
    ;; in; the header-line refresh button covers everything else.
    (treemacs-filewatch-mode 1)

    ;; Filewatch-triggered refreshes go through `treemacs--process-file-events',
    ;; a different path from `treemacs-refresh' that never fires
    ;; `treemacs-post-refresh-hook' -- so decorations (hidden root, section
    ;; gaps, bold top-level folders) would otherwise go stale after an
    ;; automatic refresh. Re-decorate after every batch of filewatch changes.
    (advice-add 'treemacs--process-file-events :after
                (lambda (&rest _) (ps/file-tree--decorate-buffer)))

    (add-hook 'treemacs-mode-hook (lambda () (display-line-numbers-mode 0)))
    (add-hook 'treemacs-mode-hook
      (lambda () (setq-local frame-title-format "File Tree")))

    ;; Keep the tree pinned at column 0. Without this, resizing the frame can
    ;; push point off-screen to the right and auto-hscroll scrolls the window
    ;; sideways, cutting off the left edge. Truncating (like the agenda does)
    ;; and disabling auto-hscroll keeps it from ever scrolling horizontally.
    (add-hook 'treemacs-mode-hook
      (lambda ()
        (setq-local truncate-lines t)
        (setq-local auto-hscroll-mode nil)))

    ;; Restore some vertical breathing room between entries. `hl-line-sticky-flag'
    ;; keeps the current-file highlight drawn while the tree window is not the
    ;; selected one; treemacs relies on the Emacs default here without setting it.
    (add-hook 'treemacs-mode-hook
      (lambda ()
        (setq-local line-spacing 2)
        (setq-local hl-line-sticky-flag t)))

    ;; treemacs toggles a node on button *press* (down-mouse-1) when its icon
    ;; is clicked; our release handler toggles again, causing an icon click to
    ;; expand-then-collapse. Suppress the press action so the release handler
    ;; does a single, consistent toggle.
    (define-key treemacs-mode-map [down-mouse-1] #'ignore)

    ;; Click a label to open it on the right (file, or org heading position
    ;; for tag nodes). Click an icon, or shift-click a label, to
    ;; expand/collapse — a no-op on leaf nodes. TAB still shows an org file's
    ;; structure.
    (define-key treemacs-mode-map [mouse-1] #'ps/file-tree-click)
    (define-key treemacs-mode-map [S-mouse-1] #'ps/file-tree-shift-click)

    ;; The right-click "Open" action (`treemacs-visit-node-no-split') errors
    ;; with "stringp, nil" on an org heading that has sub-headings. Route org
    ;; heading (tag) nodes through our own open logic, which jumps to the
    ;; heading position; everything else behaves as before.
    (define-advice treemacs-visit-node-no-split
        (:around (orig-fn &rest args) ps/file-tree-open-org-heading)
      (let ((btn (treemacs-current-button)))
        (if (and btn (memq (treemacs-button-get btn :state)
                           '(tag-node-open tag-node-closed tag-node)))
            (ps/file-tree--click-label)
          (apply orig-fn args))))

    ;; Non-intrusive expand/collapse-all, local to the tree buffer.
    (define-key treemacs-mode-map (kbd "M-+") #'ps/file-tree-expand-all)
    (define-key treemacs-mode-map (kbd "M--") #'ps/file-tree-collapse-all)

    ;; Keep the file tree visible while the agenda dispatcher is choosing a
    ;; command (org-agenda's window setup would otherwise close it).
    (with-eval-after-load 'org-agenda
      (define-advice org-agenda (:around (orig-fn &rest args) ps/file-tree-preserve)
        "Keep the file tree visible across agenda window reorganization.
When the dispatcher is shown (no command key given up front), leave the tree
in place while choosing; it is restored after the agenda is built either way."
        (let ((open (ps/file-tree-window-exists-p))
              (dispatcher (null (nth 1 args))))
          (when (and open (not dispatcher)) (ps/file-tree-hide))
          (unwind-protect (apply orig-fn args)
            (when open (ps/file-tree-show)))))

      ;; Companion to `ps/file-tree-preserve': split the selected window first
      ;; when it's the only content window, so Agenda/Calendar/Tasks keep
      ;; today's single-window convenience even though `org-agenda-window-setup'
      ;; (set to `current-window') no longer reorganizes the frame for them.
      (advice-add 'org-agenda :around #'ps/window--split-if-alone-advice)
      (advice-add 'org-todo-list :around #'ps/window--split-if-alone-advice)

      ;; ...but a *refresh* rebuilds a view that is already on screen, so the
      ;; same convenience there would add a redundant second window on every
      ;; redo.  Rebuild in place instead.
      (advice-add 'org-agenda-redo :around #'ps/window--inhibit-split-advice)

      ;; Show the agenda dispatcher (` *Agenda Commands*') as a bottom strip
      ;; without collapsing the frame.  Org's `org-display-buffer-split' calls
      ;; `delete-other-windows' before showing it; suppress that while the
      ;; chooser is up so the rest of the frame (file tree, splits) stays put.
      ;; The binding is reverted before any command runs, so the final agenda
      ;; build still goes through its configured `org-agenda-window-setup'.
      (define-advice org-agenda-get-restriction-and-command
          (:around (orig-fn &rest args) ps/agenda-dispatcher-keep-frame)
        "Show the agenda dispatcher as a bottom strip without collapsing the frame."
        (cl-letf (((symbol-function 'delete-other-windows) #'ignore))
          (apply orig-fn args)))))


  ;; Open the file tree at `my-org-base-directory' on startup, laid out per
  ;; `ps/file-tree-root-mode' and fully expanded. treemacs expands directories
  ;; asynchronously (git-status/flatten-dirs subprocesses), so re-run the
  ;; idempotent setup a few times to ride out late-arriving updates.
  (add-hook 'emacs-startup-hook
    (lambda ()
      (require 'treemacs)
      (save-selected-window
        (treemacs-select-window)
        (let ((base (treemacs-canonical-path my-org-base-directory)))
          (dolist (delay '(0 0.3 0.8 1.5 3))
            (run-with-timer delay nil #'ps/file-tree-init base))))))
#+end_src

  ;; < (category), _ (effort).  The mode line, line-number gutter, and

** Toggle horizontal/vertical split (=C-x |=)
#+begin_src emacs-lisp
  ;; Partially reuses the code from:
  ;; https://emacs.stackexchange.com/questions/5371/how-to-change-emacs-windows-from-vertical-split-to-horizontal-split
  (defun ps/window-split-toggle ()
    "Switch the two content windows between a horizontal (top/bottom) and
vertical (left/right) split, keeping their order: left<->top and
right<->bottom.  Side windows (the file tree, Claude Code) are ignored
entirely -- never counted, never touched -- via `ps/window--content-windows'."
    (interactive)
    (let ((wins (ps/window--content-windows)))
      (if (/= (length wins) 2)
          (error "Can only toggle with exactly 2 content windows!")
        ;; Order the windows geometrically (top-left first) rather than by
        ;; `window-list' order, which starts at the selected window and would
        ;; otherwise make placement depend on which pane is focused.
        (let* ((ordered (sort wins
                              (lambda (a b)
                                (let ((ea (window-edges a))
                                      (eb (window-edges b)))
                                  (if (= (nth 1 ea) (nth 1 eb))
                                      (< (nth 0 ea) (nth 0 eb))   ;; same top: leftmost first
                                    (< (nth 1 ea) (nth 1 eb))))))) ;; otherwise topmost first
               (win1 (nth 0 ordered))
               (win2 (nth 1 ordered))
               (buf1 (window-buffer win1))
               (buf2 (window-buffer win2))
               (split-func (if (window-full-height-p win1)
                               #'split-window-vertically
                             #'split-window-horizontally)))
          ;; Delete only win2 -- never `delete-other-windows', which would
          ;; also sweep away any side window (file tree, Claude Code).
          (delete-window win2)
          (select-window win1)
          (let ((new-win (funcall split-func)))
            (set-window-buffer win1 buf1)
            (set-window-buffer new-win buf2))))))

  (global-set-key (kbd "C-x |") 'ps/window-split-toggle)
#+end_src

  ;; https://emacs.stackexchange.com/questions/639/how-can-i-restart-emacs-and-preserve-my-open-buffers-and-interactive-history
  (desktop-save-mode 1)

  (add-to-list 'savehist-additional-variables 'ps/file-tree-current-set)

  (add-to-list 'savehist-additional-variables 'ps/file-tree-set-applies-to-agenda)

  ;; https://www.gnu.org/software/emacs/manual/html_node/emacs/Saving-Emacs-Sessions.html
  (setq desktop-load-locked-desktop t)

  ;; Desktop restores the whole frame layout, so any extra frame that happened
  ;; to be open when Emacs was last closed reappears on the next start, looking
  ;; like a window nobody opened.  Keep the restored size and position of the
  ;; initial frame, but drop the extra top-level frames.  Child frames (the
  ;; scrollbar pill and friends) are left alone.
  (add-hook 'desktop-after-read-hook
            (lambda ()
              (let ((keep (selected-frame)))
                (dolist (frame (frame-list))
                  (unless (or (eq frame keep) (frame-parent frame))
                    (delete-frame frame t))))))

  ;; Desktop restores each buffer's saved minor modes *after* the mode hooks
  ;; run, so a doc buffer saved with the typo checker on comes back with it on
  ;; despite the doc-buffer exclusion.  Once the whole session is restored, turn
  ;; the checker back off in every (read-only) documentation buffer.
  (add-hook 'desktop-after-read-hook
            (lambda ()
              (dolist (buf (buffer-list))
                (with-current-buffer buf
                  (when (and (derived-mode-p 'org-mode)
                             (ps/help--doc-buffer-p)
                             (bound-and-true-p jinx-mode))
                    (jinx-mode -1))))))
