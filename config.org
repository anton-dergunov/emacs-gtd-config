#+TITLE: Emacs Configuration for Productivity, Planning, "Getting Things Done" and Knowledge Management with Org Mode

This config is tangled and loaded by =init.el=. Self-contained logic lives in
=lisp/ps-*.el= modules and is ERT-tested; this file is configuration only.

If you are reading this to make it your own, the only section you usually need is
=* Settings= below — color theme, font, and every module knob live there.

* Bootstrap
** Package management
The first block carries the =-*- lexical-binding: t; -*-= cookie that the tangled
file needs. Do not move it.
#+begin_src emacs-lisp
  ;; -*- lexical-binding: t; -*-
  ;; Code above enables lexical scoping so lambdas can capture variables (needed for async code)

  ;; Disable the org-element cache before anything else.  Without this,
  ;; Org can hit "Invalid search bound (wrong side of point)" while
  ;; parsing this very file at startup — before the Org Mode section
  ;; gets to set it later.
  (setq org-element-use-cache nil)

  (require 'package)

  ;; Initialize package sources if needed
  (setq package-archives '(("melpa" . "https://melpa.org/packages/")
                           ("elpa" . "https://elpa.gnu.org/packages/")))

  ;; Refresh package contents if needed
  (unless package-archive-contents
    (package-refresh-contents))

  ;; Install use-package if missing
  (unless (package-installed-p 'use-package)
    (package-install 'use-package))

  (require 'use-package)
  (setq use-package-always-ensure t)
#+end_src

** Appearance
Set =ps/color-theme= to any installed or built-in theme; external theme packages
(solarized, doom, ef, modus) are auto-installed on first load. The Solarized
hand-tuning in =Editor & UI / Color theme= and the colour tweaks throughout this
config are applied verbatim only for =solarized-*= themes; every other theme gets
theme-relative equivalents automatically, so switching is a one-line change.
#+begin_src emacs-lisp
  ;; Pick your look here. Recommended values (auto-installed if from a package):
  ;;   Solarized (pkg):   solarized-light (default)  solarized-dark
  ;;   modus (built-in):  modus-operandi  modus-vivendi
  ;;   ef-themes (pkg):   ef-day  ef-light  ef-elea-dark  ef-winter  ef-autumn
  ;;   standard (pkg):    standard-light  standard-dark
  ;;   doric (pkg):       doric-light  doric-dark
  ;;   doom (pkg):        doom-one  doom-one-light  doom-nord  doom-dracula  doom-gruvbox
  ;;   Catppuccin (pkg):  batppuccin-latte  batppuccin-mocha  batppuccin-macchiato  batppuccin-frappe
  ;;   Tokyo Night (pkg): tokyo-night  tokyo-night-storm  tokyo-night-moon  tokyo-night-day
  ;;   Gruvbox (pkg):     gruvbox-dark-medium  gruvbox-light-medium
  ;;   Nord (pkg):        nord
  ;; Audition one without editing this file:  M-x ps/preview-theme
  (defvar ps/color-theme 'solarized-light
    "Color theme loaded by `Editor & UI / Color theme'.")

  (defvar ps/color-theme-solarized-p
    (string-prefix-p "solarized-" (symbol-name ps/color-theme))
    "Non-nil when `ps/color-theme' is a Solarized variant.
Colour tweaks across this config keep their original hand-tuned values when
this is non-nil, and fall back to theme-relative faces otherwise.")

  (defvar ps/main-font
    (cond
     ((eq system-type 'darwin)     "Monaco-14")
     ((eq system-type 'windows-nt) "Consolas-11")
     ((eq system-type 'gnu/linux)  "monospace-10")
     (t                            "default-font"))
    "Default font, applied by `Editor & UI / Font'.")

** Color theme
Loads =ps/color-theme= from Settings, auto-installing the package that provides it
when needed. The Solarized-specific tuning below only runs for =solarized-*=
themes; any other theme loads cleanly without it.
#+begin_src emacs-lisp
  ;; Theme-name prefix -> package that provides it. Built-in themes (e.g. modus-*
  ;; on Emacs 28+) install only if not already available, so listing them is harmless.
  (defvar ps/color-theme-packages
    '(("modus-"      . modus-themes)     ; accessible "default" style (built-in 28+)
      ("ef-"         . ef-themes)        ; colorful + legible (GNU ELPA)
      ("standard-"   . standard-themes)  ; faithful-to-default-Emacs, refined (GNU ELPA)
      ("doric-"      . doric-themes)     ; minimalist (GNU ELPA)
      ("doom-"       . doom-themes)      ; the popular megapack (MELPA)
      ("batppuccin-" . batppuccin)       ; Catppuccin, as 4 proper themes (MELPA)
      ("tokyo-night" . tokyo-night)      ; Tokyo Night, 4 variants (MELPA)
      ("gruvbox"     . gruvbox-theme)    ; retro groove (NonGNU ELPA / MELPA)
      ("nord"        . nord-theme)       ; arctic palette (MELPA)
      ("solarized-"  . solarized-theme)) ; default theme, kept for back-compat
    "Alist of (THEME-NAME-PREFIX . PACKAGE) for auto-installing theme packages.")

  (defun ps/load-color-theme (theme)
    "Install (if needed) and load THEME, applying Solarized tuning when relevant."
    (let* ((name (symbol-name theme))
           (pkg (cdr (seq-find (lambda (c) (string-prefix-p (car c) name))
                               ps/color-theme-packages))))
      ;; Pull in the providing package unless THEME is already available.
      (when (and pkg (not (memq theme (custom-available-themes))))
        (unless (package-installed-p pkg)
          ;; Always refresh: the cached archive contents may predate this
          ;; package's addition to MELPA/GNU ELPA, or reference a since-pruned
          ;; tarball version, both of which surface as "package unavailable".
          (package-refresh-contents)
          (package-install pkg))
        (require pkg nil t))
      ;; Solarized-only tuning (these variables don't exist for other themes).
      (when (string-prefix-p "solarized-" name)
        (setq
         solarized-distinct-fringe-background nil  ; fringe blends with background
         solarized-use-variable-pitch nil          ; keep one font for headings/titles
         solarized-high-contrast-mode-line nil     ; low-contrast modeline
         solarized-use-less-bold t                 ; less bolding
         solarized-use-more-italic t               ; more italics
         solarized-emphasize-indicators nil        ; muted git-gutter/flycheck indicators
         solarized-scale-org-headlines nil         ; don't resize org headlines
         ;; Avoid all font-size changes
         solarized-height-minus-1 1.0
         solarized-height-plus-1 1.0
         solarized-height-plus-2 1.0
         solarized-height-plus-3 1.0
         solarized-height-plus-4 1.0))
      (load-theme theme t)))

  (ps/load-color-theme ps/color-theme)

  ;; On the very first frame, Emacs may not yet know the display supports 256+
  ;; colors, so a theme's GUI-specific `default' face spec (e.g. doom themes'
  ;; `min-colors 256' background) can fail to apply, leaving the frame's
  ;; previous/default background behind. Reloading once the initial frame is
  ;; fully set up fixes the background without affecting already-correct themes.
  (add-hook 'window-setup-hook
            (lambda () (load-theme ps/color-theme t)))
#+end_src

** Live theme preview
=M-x ps/preview-theme= (also =C-c p T=) loads a theme immediately so you can
audition it; set =ps/color-theme= in Settings to make a choice permanent. Note
that the per-theme colour tweaks elsewhere in this config are only recomputed on
restart, so a previewed theme may show the Solarized greys until you restart.
#+begin_src emacs-lisp
  (defun ps/preview-theme (theme)
    "Load THEME now for preview, disabling any currently enabled themes first.
THEME may be a symbol from an as-yet-uninstalled package; it is auto-installed.
This does not persist — edit `ps/color-theme' in Settings to keep a theme."
    (interactive
     (list (intern (completing-read
                    "Preview theme: "
                    (mapcar #'symbol-name (custom-available-themes))))))
    (mapc #'disable-theme custom-enabled-themes)
    (ps/load-color-theme theme)
    (message "Previewing %s — set ps/color-theme in config.org to keep it" theme))
#+end_src

** Appearance
*** Emacs 30 compatibility shims
=internal--without-restriction= was removed from some Emacs 30 builds, but
org-modern's compiled code still calls it as a function, passing a thunk
(and optionally a label). Provide that function, not a macro: a macro's
function cell can't satisfy a runtime function call.
#+begin_src emacs-lisp
  (unless (fboundp 'internal--without-restriction)
    (defun internal--without-restriction (thunk &optional _label)
      (save-restriction
        (widen)
        (funcall thunk))))
#+end_src

  ;; Apply a faded color to Org's native metadata faces. Solarized keeps the
  ;; original light gray; other themes derive a dim colour from `shadow'.
  (defun ps/org-apply-metadata-fade ()
    "Fade all #+ metadata lines natively."
    (let ((color (if ps/color-theme-solarized-p
                     "gray85"
                   (face-foreground 'shadow nil t))))
      ;; Changes faces like `#+TIMEZONE:` and `#+PROPERTY:`
      (set-face-attribute 'org-meta-line nil :foreground color)
      ;; Changes the `#+TITLE:` label specifically
      (set-face-attribute 'org-document-info-keyword nil :foreground color)
      ;; Changes the actual value (e.g., "Interviews") next to TITLE
      (set-face-attribute 'org-document-info nil :foreground color)))

  ;; Run it on mode start
  (add-hook 'org-mode-hook #'ps/org-apply-metadata-fade)
#+end_src

** =treemacs= file tree
A tree-style sidebar for browsing the Org GTD tree.
The ignore list and category icons live in =lisp/ps-file-tree.el= and
=lisp/ps-file-tree-icons.el=.
#+begin_src emacs-lisp
  (use-package treemacs
    :defer t
    :commands (treemacs treemacs-select-window)
    :config
    (ps/file-tree-setup-ignore)
    (if ps/file-tree-use-custom-icons
        (ps/file-tree-icons-apply ps/file-tree-icon-dirs my-org-base-directory)
      (treemacs-load-theme "Default"))

    ;; Root project labels (Areas/Current/Vision) should look like ordinary
    ;; file-tree entries (not underlined or oversized like the default
    ;; hyperlink-style root face), but bold to stand out as section headers.
    (set-face-attribute 'treemacs-root-face nil
      :inherit 'treemacs-file-face
      :height 1.0
      :weight 'bold
      :underline nil)

    ;; Narrower than treemacs's 35-column default.
    (setq treemacs-width 25)

    ;; Allow the tree window to be resized like any other window.
    (setq treemacs-width-is-initially-locked nil)

    ;; More compressed indentation per level (default is 2).
    (setq treemacs-indentation 1)

    ;; Hide the .org extension and add breathing room after icons.
    (setq treemacs-file-name-transformer #'ps/file-tree-transform-file-name)
    (setq treemacs-directory-name-transformer #'ps/file-tree-transform-dir-name)

    ;; Never follow the current buffer/file - the tree stays where it is.
    (treemacs-follow-mode -1)
    (treemacs-tag-follow-mode -1)

    ;; Add a small gap above the first tree line without a visible separator.
    (add-hook 'treemacs-mode-hook
      (lambda ()
        (setq header-line-format (propertize " " 'display '(space :height (3))))
        (face-remap-add-relative 'header-line
          :box nil :underline nil :overline nil :background 'unspecified :inherit nil)))
    (add-hook 'treemacs-mode-hook (lambda () (display-line-numbers-mode 0)))
    (add-hook 'treemacs-mode-hook
      (lambda () (setq-local frame-title-format "File Tree")))

    ;; Restore some vertical breathing room between entries.
    (add-hook 'treemacs-mode-hook (lambda () (setq-local line-spacing 2)))

    ;; Single click opens files / toggles folders, alongside the default
    ;; double-click bindings; TAB still shows an org file's structure.
    (define-key treemacs-mode-map [mouse-1] #'treemacs-single-click-expand-action)

    ;; Non-intrusive expand/collapse-all, local to the tree buffer.
    (define-key treemacs-mode-map (kbd "M-+") #'ps/file-tree-expand-all)
    (define-key treemacs-mode-map (kbd "M--") #'ps/file-tree-collapse-all)

    ;; Keep the file tree visible across agenda window reorganization
    ;; (org-agenda's default window setup would otherwise close it).
    (with-eval-after-load 'org-agenda
      (define-advice org-agenda (:around (orig-fn &rest args) ps/file-tree-preserve)
        "Keep the file tree visible across agenda window reorganization."
        (let ((open (ps/file-tree-window-exists-p)))
          (when open (ps/file-tree-hide))
          (apply orig-fn args)
          (when open (ps/file-tree-show))))))

  ;; Open the file tree at `my-org-base-directory' on startup, one project per
  ;; immediate subdirectory, fully expanded. treemacs expands directories
  ;; asynchronously (git-status/flatten-dirs subprocesses), so re-run the
  ;; idempotent setup a few times to ride out late-arriving updates.
  (add-hook 'emacs-startup-hook
    (lambda ()
      (require 'treemacs)
      (save-selected-window
        (treemacs-select-window)
        (let ((base (treemacs-canonical-path my-org-base-directory)))
          (dolist (delay '(0 0.3 0.8 1.5 3))
            (run-with-timer delay nil #'ps/file-tree-init base))))))
#+end_src

** Toggle horizontal/vertical split (=C-x |=)
#+begin_src emacs-lisp
  ;; Partially reuses the code from:
  ;; https://emacs.stackexchange.com/questions/5371/how-to-change-emacs-windows-from-vertical-split-to-horizontal-split
  (defun ps/window-split-toggle ()
    "Toggle between horizontal and vertical split with exactly two windows, preserving buffer order."
    (interactive)
    ;; Check if the file tree is open
    (let ((tree-open (ps/file-tree-window-exists-p)))
      (when tree-open
        (ps/file-tree-hide))  ;; Hide it temporarily

      ;; Check if there are exactly two windows
      (if (/= (length (window-list)) 2)
          (error "Can only toggle with exactly 2 windows!")
        (let* ((win1 (nth 0 (window-list)))
               (win2 (nth 1 (window-list)))
               (buf1 (window-buffer win1))
               (buf2 (window-buffer win2))
               (split-func (if (window-full-height-p)
                               #'split-window-vertically
                             #'split-window-horizontally)))
          (delete-other-windows win1)
          (let ((new-win (funcall split-func)))
            (set-window-buffer win1 buf1)
            (set-window-buffer new-win buf2))))

      ;; Reopen the file tree if it was open before
      (when tree-open
        (ps/file-tree-show))))

  (global-set-key (kbd "C-x |") 'ps/window-split-toggle)
#+end_src

** Productivity menu + prefix keys
Built with =easy-menu-define= (built-in =easymenu=). Items are grouped into
submenus; the "Git Sync Enabled" entry is a toggle whose tick reflects

("Productivity") prefix — Emacs reserves =C-c LETTER= bindings for the user

** Automatic git sync for org files
Logic lives in =lisp/ps-git-sync.el=. The interval is set in =* Settings / Git sync=
above. =ps/git-sync-resume= clears the paused state after you resolve a merge
conflict.

Set the =PS_GIT_SYNC_DISABLE= environment variable (done by
=scripts/run_emacs_dev.sh=) to skip starting the sync timer entirely. This is
more robust than relying on =ps/git-sync-paused=, which can be cleared during
a session (by =ps/git-sync-resume=, a menu toggle, or conflict handling).
#+begin_src emacs-lisp
  (unless (getenv "PS_GIT_SYNC_DISABLE")
    (ps/git-sync-start my-org-base-directory))
#+end_src
